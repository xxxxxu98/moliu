# 大纲字段应用于续写的完整优化方案

> 文档版本：v1.0 | 日期：2026-05-31 | 状态：规划中

---

## 一、现状问题总览

### 1.1 数据流全景

```
大纲生成 (unified-generator.ts)
    ↓
plotOutline[] (存储在 project.plotOutline)
    ↓
┌──────────────────────────────┐
│  composables 层              │
│  useChapterWriter.ts         │ ← 智能续写（单章）
│  useBatchWriter.ts           │ ← 批量续写
└──────────────────────────────┘
    ↓ 构建 context 对象
ProjectContext
    ↓
┌──────────────────────────────┐
│  services/ai 层             │
│  unified.service.ts          │
│  base.service.ts            │
│  PromptBuilder.buildContinuePrompt()
└──────────────────────────────┘
    ↓
AI Prompt (userPrompt + systemPrompt)
```

### 1.2 问题清单

| # | 问题 | 影响程度 | 涉及文件 |
|---|---|---|---|
| P1 | `chapterType`、`hookType`、`pacingStrategy` 等大纲字段**未被拼接**进 `currentChapterOutline` | 🔴 高 | `useChapterWriter.ts`、`useBatchWriter.ts` |
| P2 | `emotionGoal`、`conflictDesign`、`coolPointDesign`、`storyLines`、`coreSellingPoints` **完全未传入** `ProjectContext` | 🔴 高 | `base.service.ts`、`unified.service.ts` |
| P3 | UI 选择的 `writingStyle` **未传入** `ProjectContext` 也**未进入** prompt | 🟡 中 | `AIPanel.vue`、`base.service.ts` |
| P4 | `extractChapterOutlineFromPlot` 和 `buildFullOutlineString` **代码重复**，字段拼接逻辑分散 | 🟡 中 | `useChapterWriter.ts`、`useBatchWriter.ts` |
| P5 | `keyEvents`、`timeSpan` 在 `fullOutline` 中有拼接，但**不在** `currentChapterOutline` 中（单章任务） | 🟡 中 | 同上 |
| P6 | `ProjectContext` 类型与实际使用存在**不一致**——`aiParams` 直接传了 `relatedForeshadows` 但 `ProjectContext` 定义了却没用 | 🟢 低 | `base.service.ts`、`useBatchWriter.ts` |

---

## 二、目标

1. **全部大纲字段进入 prompt**——章节设计层的每个字段都能影响 AI 输出
2. **消除重复代码**——统一大纲提取逻辑，避免两处维护
3. **类型安全**——`ProjectContext` 完整覆盖所有可用字段
4. **向后兼容**——不破坏现有流程，已有的字段继续生效

import type { PlotNode } from '@/types/project';

/**
 * 从 PlotNode 提取完整章节上下文
 */
export function extractChapterContext(
  plotOutline: PlotNode[],
  chapterId: string,
  chapterTitle?: string
): ChapterOutlineContext | null {
  // 优先按 chapterId 匹配
  let node = plotOutline.find(p => p.chapterId === chapterId);
  // 降级按 id 匹配
  if (!node) node = plotOutline.find(p => p.id === chapterId);

  if (!node) return null;

  return {
    title: node.title || chapterTitle || '未命名',
    description: node.description || '',
    orderIndex: node.orderIndex ?? 0,

    // 结构化节点
    CBN: node.CBN,
    CPNs: node.CPNs,
    CEN: node.CEN,
    mustCover: node.mustCover,
    forbiddenZones: node.forbiddenZones,

    // 章节写作策略
    chapterType: node.chapterType,
    hookType: node.hookType,
    pacingStrategy: node.pacingStrategy,
    timeSpan: node.timeSpan,
    keyEvents: node.keyEvents,
    isClimax: node.isClimax,
    expectedCoolPoints: node.expectedCoolPoints,
  };
}

/**
 * 将 ChapterOutlineContext 拼接为 AI 可读的 prompt 文本
 */
export function buildChapterOutlineText(
  ctx: ChapterOutlineContext,
  includeStrategyFields: boolean = true
): string {
  const parts: string[] = [];

  // 基础大纲描述
  if (ctx.description) {
    parts.push(ctx.description);
  }

  // 结构化节点
  if (ctx.CBN) parts.push(`【章节起点 CBN】${ctx.CBN}`);
  if (ctx.CPNs?.length) {
    parts.push(`【推进节点 CPNs】\n  ${ctx.CPNs.map((c, i) => `${i + 1}. ${c}`).join('\n  ')}`);
  }
  if (ctx.CEN) parts.push(`【章节终点 CEN】${ctx.CEN}`);
  if (ctx.mustCover?.length) parts.push(`【必须覆盖】${ctx.mustCover.join('、')}`);
  if (ctx.forbiddenZones?.length) parts.push(`【禁区】${ctx.forbiddenZones.join('、')}`);

  // 章节写作策略（新增）
  if (includeStrategyFields) {
    if (ctx.chapterType) {
      const typeLabels: Record<string, string> = {
        world_intro: '世界观介绍',
        character_intro: '人物登场',
        plot_setup: '情节铺陈',
        conflict: '冲突展开',
        climax: '高潮',
        resolution: '冲突解决',
        transitional: '过渡章节',
        ending: '结尾收束',
        normal: '普通章节',
      };
      parts.push(`【章节类型】${typeLabels[ctx.chapterType] || ctx.chapterType}`);
    }

    if (ctx.hookType) {
      const hookLabels: Record<string, string> = {
        sudden_reveal: '突然揭示',
        urgent_crisis: '紧急危机',
        unfinished_action: '未完成动作',
        identity_reveal: '身份反转',
        tough_choice: '两难抉择',
        mysterious_item: '神秘物品',
        countdown: '倒计时',
        promise_threat: '承诺/威胁',
        strange_disappear: '神秘消失',
        hidden_meaning: '隐含深意',
        imagery: '意象留白',
        echo: '首尾呼应',
        blank: '悬念留白',
      };
      parts.push(`【章尾钩子】${hookLabels[ctx.hookType] || ctx.hookType}`);
    }

    if (ctx.pacingStrategy) {
      const paceLabels: Record<string, string> = {
        build_up: '蓄势铺垫（慢节奏）',
        confront: '对峙冲突（中节奏）',
        release: '释放爆发（快节奏）',
        normal: '正常节奏',
      };
      parts.push(`【节奏策略】${paceLabels[ctx.pacingStrategy] || ctx.pacingStrategy}`);
    }

    if (ctx.timeSpan) {
      parts.push(`【章节时长】${ctx.timeSpan}`);
    }

    if (ctx.keyEvents?.length) {
      parts.push(`【关键事件】${ctx.keyEvents.join(' → ')}`);
    }

    if (ctx.isClimax) {
      parts.push('【重要】本章为高潮章节，需强化冲突和情感张力');
    }
  }

  return parts.join('\n');
}

/**
 * 构建完整大纲文本（所有章节）
 */
export function buildFullOutlineText(plotOutline: PlotNode[]): string {
  const chapters = plotOutline
    .filter(p => p.type === 'chapter')
    .sort((a, b) => a.orderIndex - b.orderIndex);

  if (chapters.length === 0) return '';

  return chapters.map((node, i) => {
    const num = i + 1;
    const title = node.title || `第${num}章`;
    const desc = node.description || '（暂无大纲）';
    const ctx = extractChapterContext(plotOutline, node.id, title);
    const outlineText = ctx ? buildChapterOutlineText(ctx, true) : desc;
    const isClimax = node.isClimax ? ' ⭐高潮' : '';
    return `【第${num}章】${title}${isClimax}\n${outlineText}`;
  }).join('\n\n');
}

/**
 * 构建增强设计 prompt 段落
 */
export function buildEnhancedDesignPrompt(ctx: EnhancedProjectContext): string {
  const sections: string[] = [];

  // 情绪目标
  if (ctx.emotionGoal) {
    const eg = ctx.emotionGoal;
    sections.push(`## 【情绪目标】
- 核心情绪：${eg.primary}
${eg.secondary ? `- 次要情绪：${eg.secondary}` : ''}
- 情绪弧线：${eg.arc}
- 波动间隔：每 ${eg.density} 字
${eg.highPoints?.length ? `- 情绪高点章节：第 ${eg.highPoints.join('、')} 章` : ''}
${eg.lowPoints?.length ? `- 情绪低点章节：第 ${eg.lowPoints.join('、')} 章` : ''}
请根据当前章节的情绪定位，合理安排情感铺垫和爆发。`);
  }

  // 矛盾设计
  if (ctx.conflictDesign) {
    const cd = ctx.conflictDesign;
    sections.push(`## 【矛盾设计】
- 冲突来源：${cd.source}
- 矛盾递进：${cd.escalation.map(e => `${e.level}级：${e.name}`).join(' → ')}
- 主要冲突：
${cd.majorConflicts.map(c => `  · ${c.title}（${c.type}级）`).join('\n')}
请在章节中推进或揭示相关矛盾冲突。`);
  }

  // 爽点设计
  if (ctx.coolPointDesign) {
    const cp = ctx.coolPointDesign;
    const patterns = cp.patterns?.map(p => {
      const labels: Record<string, string> = {
        'face-slapping': '打脸',
        'show-off': '装逼',
        'identity-reveal': '身份揭秘',
        'growth': '成长突破',
        'rescue': '英雄救美',
        'treasure': '寻宝获宝',
        'breakthrough': '境界突破',
        'romance': '甜蜜恋爱',
        'revenge': '复仇快感',
        'mystery-reveal': '谜题揭开',
        'comedy': '搞笑逗比',
        'justice': '伸张正义',
      };
      return labels[p] || p;
    }).join('、') || '';

    sections.push(`## 【爽点设计】
- 爽点类型：${patterns}
- 爽点密度：微爽每 ${cp.density?.micro} 字 / 小爽每 ${cp.density?.small} 字 / 大爽每 ${cp.density?.big} 字
${cp.arranged?.length ? `- 已安排爽点：\n${cp.arranged.map(a => `  · 第${a.chapter}章：${a.type} - ${a.description}`).join('\n')}` : ''}
请在章节中安排适当的爽点节奏。`);
  }

  // 八条故事线
  if (ctx.storyLines) {
    const sl = ctx.storyLines;
    sections.push(`## 【八条故事线】
- 地图线：${sl.map?.current || '（暂无）'} ${sl.map?.planned?.length ? `（规划：${sl.map.planned.join('→')}）` : ''}
- 阵营线：${sl.faction?.planned?.join('、') || '（暂无）'}
- 人物线：${sl.character?.planned?.map(p => p.role).join('、') || '（暂无）'}
- 金手指线：${sl.goldenfinger?.type || '（暂无）'} - 当前阶段：${sl.goldenfinger?.currentStage || 0}
- 世界观线：${sl.worldRules?.revealed?.join('、') || '（暂无）'} 待揭示：${sl.worldRules?.pending?.join('、') || '无'}
- 矛盾线：${sl.conflict?.activeConflict || '（暂无）'}
- 收集线：${sl.collection?.progress?.filter(p => p.acquired).map(p => p.item).join('、') || '（暂无）'}
- 感情线：${sl.romance?.currentStage || '（暂无）'}
请确保章节内容推进相关故事线的发展。`);
  }

  // 核心卖点
  if (ctx.coreSellingPoints?.length) {
    sections.push(`## 【核心卖点】
${ctx.coreSellingPoints.map(p => `- ${p.name}：${p.description}`).join('\n')}
请确保章节内容体现和强化这些核心卖点。`);
  }

  return sections.join('\n\n');
}

## 六、风险与注意事项

### 6.1 Prompt 长度膨胀
增加大量字段可能导致 prompt 超出 Token 限制。

**缓解措施：**
- 对可选字段做**守卫检查**，无数据则不输出
- 设置 Token 上限，超出时裁剪 `recentChaptersFullText`（`PromptBuilder` 已有 `truncateToTokenLimit` 方法）
- 优先传递当前章节相关字段，历史信息可缩短

### 6.2 兼容性
新增 prompt 段落不应破坏现有生成结果。

**缓解措施：**
- 新增段落放在 `userPrompt` 的**末尾**（在 `# 任务要求` 之后）
- 所有新增内容均为**补充指导**，不覆盖原有核心指令
- 通过功能开关控制（`useEnhancedOutlineFields` 配置项）

### 6.3 AI 遵从度
AI 可能不完全遵从新增的所有字段。

**缓解措施：**
- 在 prompt 中使用**强制标记**（如【重要】【强制】【必须】）
- 提供**具体示例**，而非抽象描述
- 监控初期使用效果，逐步优化措辞
